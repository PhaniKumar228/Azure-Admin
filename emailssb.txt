SELECT
    dp.name AS DatabaseUser,
    sp.name AS LoginName,
    'ALTER USER [' + dp.name + '] WITH LOGIN = [' + sp.name + '];' AS FixCommand
FROM sys.database_principals dp
INNER JOIN sys.server_principals sp
    ON dp.name = sp.name
WHERE dp.type = 'S'
AND dp.sid <> sp.sid
AND dp.principal_id > 4
ORDER BY dp.name;
