# Folder to scan
$folderPath = "C:\InputFiles"

# Output file
$outputFile = "C:\Output\PathExtraction.csv"

# Initialize result array
$result = @()

# Get all .config and .ini files recursively
$files = Get-ChildItem -Path $folderPath -Include *.config, *.ini -Recurse -File

foreach ($file in $files) {

    # Read file line by line
    $lines = Get-Content $file.FullName -ErrorAction SilentlyContinue

    foreach ($line in $lines) {

        # Match any path containing \ or \\ (local or UNC)
        if ($line -match '(\\\\[^\s"=]+|[A-Za-z]:\\[^\s"=]+)') {

            # Extract all matches
            $matches = [regex]::Matches($line, '(\\\\[^\s"=]+|[A-Za-z]:\\[^\s"=]+)')

            foreach ($match in $matches) {

                $result += [PSCustomObject]@{
                    FileName = $file.Name
                    Path     = $match.Value
                }
            }
        }
    }
}

# Export to CSV
$result | Sort-Object FileName | Export-Csv -Path $outputFile -NoTypeInformation

Write-Host "✅ Extraction completed. Output saved to $outputFile"
