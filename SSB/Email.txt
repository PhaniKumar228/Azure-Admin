SELECT
    s.name AS SessionName,
    t.target_name,
    CAST(t.target_data AS XML) AS TargetData
FROM sys.server_event_sessions s
JOIN sys.server_event_session_targets t
    ON s.event_session_id = t.event_session_id
WHERE t.target_name = 'event_file';


ALTER EVENT SESSION [MySession]
ON SERVER
STATE = STOP;
GO

DROP EVENT SESSION [MySession]
ON SERVER;
GO

SELECT name
FROM sys.server_event_sessions
ORDER BY name;
