@echo off
setlocal enabledelayedexpansion

set TXTFILE=Users.txt
set CSVFILE=Users.csv
set REPORT=ComparisonReport.txt

echo User Comparison Report > %REPORT%
echo ====================== >> %REPORT%
echo. >> %REPORT%

for /f "tokens=1,2 delims=," %%A in (%TXTFILE%) do (

    set USERID=%%A
    set FOUND=N

    for /f "skip=1 tokens=1-4 delims=," %%i in (%CSVFILE%) do (

        if /I "%%i"=="!USERID!" (
            set FOUND=Y

            echo User Found : %%i >> %REPORT%
            echo UserName   : %%j >> %REPORT%
            echo Group      : %%k >> %REPORT%
            echo Status     : %%l >> %REPORT%

            if "%%l"=="Inactive" (
                echo WARNING: User is Inactive >> %REPORT%
            )

            echo -------------------------------- >> %REPORT%
        )
    )

    if "!FOUND!"=="N" (
        echo MISSING USER : !USERID! >> %REPORT%
        echo -------------------------------- >> %REPORT%
    )
)

echo Report generated: %REPORT%
pause
`
