-- ================================================================
-- Generate INSERT statements for [IVSCPAM].[dbo].[mhfshdr]
-- FIX: Uses CHAR(39) = ' and CHAR(44) = , instead of escaped quotes
-- 1. Run on SOURCE server in SSMS
-- 2. Copy all results rows
-- 3. Paste and run on TARGET server
-- ================================================================

DECLARE @TableName  NVARCHAR(256) = 'mhfshdr'
DECLARE @SchemaName NVARCHAR(256) = 'dbo'
DECLARE @SQL        NVARCHAR(MAX)  = ''
DECLARE @ColList    NVARCHAR(MAX)  = ''
DECLARE @ValList    NVARCHAR(MAX)  = ''

-- Step 1: Build column list
SELECT @ColList = @ColList + QUOTENAME(COLUMN_NAME) + ','
FROM   INFORMATION_SCHEMA.COLUMNS
WHERE  TABLE_NAME   = @TableName
  AND  TABLE_SCHEMA = @SchemaName
ORDER BY ORDINAL_POSITION

-- Remove trailing comma
SET @ColList = LEFT(@ColList, LEN(@ColList) - 1)

-- Step 2: Build value expressions per data type
--   CHAR(39) = single quote '
--   CHAR(44) = comma ,
--   No nested quote escaping needed!
SELECT @ValList = @ValList +
    CASE
        -- Numeric types: cast to string, no quoting
        WHEN DATA_TYPE IN (
            'int','bigint','smallint','tinyint','bit',
            'decimal','numeric','float','real','money','smallmoney'
        )
        THEN 'ISNULL(CAST(' + QUOTENAME(COLUMN_NAME) + ' AS NVARCHAR(MAX)),''NULL'')'

        -- Date/time types: wrap in single quotes via CHAR(39)
        WHEN DATA_TYPE IN (
            'date','datetime','datetime2',
            'smalldatetime','time','datetimeoffset'
        )
        THEN 'ISNULL(CHAR(39)+CONVERT(NVARCHAR(50),'
               + QUOTENAME(COLUMN_NAME)
               + ',121)+CHAR(39),''NULL'')'

        -- All other types (varchar, nvarchar, char, uniqueidentifier, xml, etc.)
        -- Wrap in CHAR(39) and escape any embedded single quotes
        ELSE
            'ISNULL(CHAR(39)+REPLACE(CAST('
               + QUOTENAME(COLUMN_NAME)
               + ' AS NVARCHAR(MAX)),CHAR(39),CHAR(39)+CHAR(39))+CHAR(39),''NULL'')'
    END + '+CHAR(44)+'   -- CHAR(44) = comma separator between values
FROM   INFORMATION_SCHEMA.COLUMNS
WHERE  TABLE_NAME   = @TableName
  AND  TABLE_SCHEMA = @SchemaName
ORDER BY ORDINAL_POSITION

-- Remove trailing +CHAR(44)+ (10 characters)
SET @ValList = LEFT(@ValList, LEN(@ValList) - 10)

-- Step 3: Assemble the full dynamic SELECT statement
SET @SQL =
      'SELECT ''INSERT INTO [IVSCPAM].[dbo].[mhfshdr] ('
    + @ColList
    + ') VALUES ('' + '
    + @ValList
    + ' + '')'' FROM [IVSCPAM].[dbo].[mhfshdr]'

-- Uncomment to inspect the generated SQL before running:
-- PRINT @SQL

EXEC sp_executesql @SQL
